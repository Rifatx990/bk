  <script>
    document.addEventListener('DOMContentLoaded', function() {
      // Set the initial countdown time (5 minutes in seconds)
      let timeLeft = 5 * 60; // 5 minutes
      const countdownTimer = document.getElementById('countdown-timer');
      const countdownMinutes = document.getElementById('countdown-minutes');
      const downloadAllBtn = document.getElementById('downloadAllBtn');
      
      // Format time to MM:SS
      function formatTime(seconds) {
        const minutes = Math.floor(seconds / 60);
        const remainingSeconds = seconds % 60;
        return `${minutes.toString().padStart(2, '0')}:${remainingSeconds.toString().padStart(2, '0')}`;
      }
      
      // Update the countdown timer every second
      const countdownInterval = setInterval(function() {
        timeLeft--;
        
        // Update timer display
        countdownTimer.textContent = formatTime(timeLeft);
        countdownMinutes.textContent = Math.ceil(timeLeft / 60);
        
        // Change color when time is running low
        if (timeLeft <= 60) {
          countdownTimer.style.color = 'var(--danger-color)';
        } else if (timeLeft <= 120) {
          countdownTimer.style.color = 'var(--warning-color)';
        }
        
        // When time runs out
        if (timeLeft <= 0) {
          clearInterval(countdownInterval);
          countdownTimer.textContent = '00:00';
          
          // Show expiration message
          const timerSection = document.querySelector('.timer-section');
          timerSection.innerHTML = `
            <div class="text-center">
              <i class="bi bi-exclamation-triangle-fill text-danger fs-1"></i>
              <h4 class="mt-3 text-danger">Files Expired</h4>
              <p class="mb-0">These download links have expired for security reasons.</p>
              <p>Please upload your document again to generate new files.</p>
              <a href="/" class="btn btn-primary mt-2">
                <i class="bi bi-upload me-2"></i> Process New Document
              </a>
            </div>
          `;
          
          // Disable download buttons
          document.querySelectorAll('.btn-download').forEach(btn => {
            btn.classList.add('disabled');
            btn.setAttribute('aria-disabled', 'true');
            btn.style.opacity = '0.6';
            btn.style.pointerEvents = 'none';
          });
          
          // Show alert
          const alertDiv = document.querySelector('.alert');
          alertDiv.innerHTML = `
            <i class="bi bi-exclamation-triangle-fill me-3 fs-4 text-danger"></i>
            <div>
              <strong>Files Expired:</strong> These download links have been deactivated for security reasons. 
              Please upload your document again to generate new files.
            </div>
          `;
          alertDiv.classList.remove('alert-info');
          alertDiv.classList.add('alert-danger');
        }
      }, 1000);
      
      // Download All Button functionality
      downloadAllBtn.addEventListener('click', function() {
        // In a real application, this would trigger a ZIP download
        // For demo purposes, we'll simulate the action
        
        // Show loading state
        const originalText = downloadAllBtn.innerHTML;
        downloadAllBtn.innerHTML = '<i class="bi bi-hourglass-split me-2"></i> Preparing ZIP...';
        downloadAllBtn.disabled = true;
        
        // Simulate ZIP preparation
        setTimeout(() => {
          alert('In a real application, this would download a ZIP file containing both CSV and PDF files.');
          
          // Reset button
          downloadAllBtn.innerHTML = originalText;
          downloadAllBtn.disabled = false;
        }, 1500);
      });
      
      // Add download tracking
      document.querySelectorAll('.btn-download').forEach(link => {
        link.addEventListener('click', function() {
          const fileType = this.classList.contains('pdf') ? 'PDF' : 'CSV';
          console.log(`Download initiated: ${fileType} file`);
          
          // In a real app, you might send analytics here
          // Example: fetch('/track-download', { method: 'POST', body: JSON.stringify({fileType}) });
        });
      });
    });
  </script>
